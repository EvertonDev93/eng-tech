-- Outliers (Cidade exterior com código Brasil)

DELETE FROM estabelecimentos
WHERE uf = 'EX'
AND pais = '105';


-- Deletar linhas de estabelecimentos onde há ausencia de cnpj

delete FROM estabelecimentos
WHERE (cnpj_basico IS NULL OR TRIM(cnpj_basico) = '')
AND (cnpj_ordem IS NULL OR TRIM(cnpj_ordem) = '')
AND (cnpj_dv IS NULL OR TRIM(cnpj_dv) = '');


-- Coluna indicando se a empresa foi aberta a mais de 5 anos
ALTER TABLE estabelecimentos
ADD COLUMN empresa_mais_5_anos BOOLEAN;

UPDATE estabelecimentos
SET empresa_mais_5_anos =
    data_inicio_atividade <= DATE '2025-11-30' - INTERVAL '5 years';


-- JOIN DO CNPJ BASICO ENTRE AS TABELAS
-- INNER JOIN TRAZ SOMENTE SE EXISTIR EM TODAS AS TABELAS
SELECT *
FROM estabelecimentos e
INNER JOIN empresas emp
    ON e.cnpj_basico = emp.cnpj_basico
INNER JOIN socios s
    ON e.cnpj_basico = s.cnpj_basico
INNER JOIN simples si
    ON e.cnpj_basico = si.cnpj_basico;


-- FULL OUTER JOIN TRAZ TUDO, INCLUSIVE REGISTROS ORFAOS SEM CORRESPONDENCIA
SELECT *
FROM estabelecimentos e
FULL OUTER JOIN empresas emp
    ON e.cnpj_basico = emp.cnpj_basico
FULL OUTER JOIN socios s
    ON COALESCE(e.cnpj_basico, emp.cnpj_basico) = s.cnpj_basico
FULL OUTER JOIN simples si
    ON COALESCE(e.cnpj_basico, emp.cnpj_basico, s.cnpj_basico) = si.cnpj_basico;
